\question{Exprimer sous forme canonique la fonction de transfert en boucle fermée $\dfrac{\beta(p)}{\beta_{c}(p)}$.}

\ifprof
\begin{corrige}
On décale le point de prélèvement vers la droite. 
On a alors la première boucle imbriquée : $F_1(p) = \dfrac{\dfrac{1}{\left( \indice{I}{1y} + \indice{B}{V}\right)p^2}}{1+\dfrac{1}{\left( \indice{I}{1y} + \indice{B}{V}\right)p^2}Mgh_2}$
$= \dfrac{1}{\left( \indice{I}{1y} + \indice{B}{V}\right)p^2+Mgh_2}$.

Pour la deuxième boucle, 

$F_2(p) = \dfrac{\dfrac{\indice{A}{V}\indice{\omega}{v}p} {\indice{B}{V} p^2}F_1(p)}{1-\dfrac{\indice{A}{V}\indice{\omega}{v}p} {\indice{B}{V} p^2}F_1(p) \indice{A}{V}\indice{\omega}{v}p}$
$= \dfrac{\indice{A}{V}\indice{\omega}{v}p F_1(p)}{\indice{B}{V} p^2-\indice{A}{V}^2 \indice{\omega}{v}^2 p^2  F_1(p) }$

$= \dfrac{\indice{A}{V}\indice{\omega}{v}p }{\left(\left( \indice{I}{1y} + \indice{B}{V}\right)p^2+Mgh_2\right)\indice{B}{V} p^2-\indice{A}{V}^2 \indice{\omega}{v}^2 p^2 }$


Pour la troisième boucle, 
$ F_3(p) = \dfrac{\indice{K}{cor}F_2(p)}{1+\indice{K}{cor}F_2(p)}$.

Ainsi, 
$ F_3(p) = \dfrac{\indice{K}{cor}\indice{A}{V}\indice{\omega}{v}p }{\left(\left( \indice{I}{1y} + \indice{B}{V}\right)p^2+Mgh_2\right)\indice{B}{V} p^2-\indice{A}{V}^2 \indice{\omega}{v}^2 p^2+\indice{K}{cor} \indice{A}{V}\indice{\omega}{v}p} $

$ F_3(p) = \dfrac{\indice{K}{cor}\indice{A}{V}\indice{\omega}{v} }{\left(\left( \indice{I}{1y} + \indice{B}{V}\right)p^2+Mgh_2\right)\indice{B}{V} p-\indice{A}{V}^2 \indice{\omega}{v}^2 p+\indice{K}{cor} \indice{A}{V}\indice{\omega}{v}} $
$= \dfrac{\indice{K}{cor}\indice{A}{V}\indice{\omega}{v} }{\left( \indice{I}{1y} + \indice{B}{V}\right)\indice{B}{V} p^3+Mgh_2\indice{B}{V} p-\indice{A}{V}^2 \indice{\omega}{v}^2 p+\indice{K}{cor} \indice{A}{V}\indice{\omega}{v}} $


$= \dfrac{1 }{
\dfrac{\left( \indice{I}{1y} + \indice{B}{V}\right)\indice{B}{V}}
{\indice{K}{cor} \indice{A}{V}\indice{\omega}{v}} p^3
+\dfrac{Mgh_2\indice{B}{V}  -\indice{A}{V}^2 \indice{\omega}{v}^2 }
{\indice{K}{cor} \indice{A}{V}\indice{\omega}{v}} p +1}
 $

\end{corrige}

\begin{question}
Établir l'expression de la fonction de transfert en boucle fermée $H(p)$ et la mettre sous la forme :
$$
H(p)=\frac{\beta(p)}{\beta_{c}(p)}=\frac{A_{V} \omega_{v}}{A\left(k_{1}, k_{3}\right)+B\left(k_{2}\right) p-C\left(k_{3}\right) p^{2}-D p^{3}}
$$

Préciser les termes $A, B, C$ et $D$ en fonction de $k_{1}, k_{2}, k_{3}$ et des paramètres du système.\\
\end{question}
\ifprof
\begin{corrige}
On déplace les points de prélèvement à droite (ce qui rajoute des facteurs $p$ dans les retours). 
On permute le comparteur 3 et le comparateur 4. 

L'ensemble des retours << du haut >> est équivalent à $k_1 +p k_2 - p\indice{A}{V}\indice{\omega}{v}$.

On a une première FTBF 

$F_1(p) =\dfrac{\dfrac{1}{\indice{B}{V}}\dfrac{1}{p}}{1+\indice{k}{3}\dfrac{1}{\indice{B}{V}}\dfrac{1}{p}}
=\dfrac{1}{\indice{B}{V} p + k_3}$. 

et 

$F_2(p) =\dfrac{\dfrac{1}{\indice{I}{1y} +\indice{B}{V}} \dfrac{1}{p^2} }{1+\dfrac{1}{\indice{I}{1y}+\indice{B}{V}} \dfrac{1}{p^2} Mgh_2}$
$=\dfrac{1}{\left(\indice{I}{1y} +\indice{B}{V} \right)p^2- Mgh_2}$. 

On a donc dans la chaîne directe : $-F_1(p) \indice{A}{V}\indice{\omega}{V} F_2(p)$.

Au final, $H(p) = \dfrac{- \indice{A}{V}\indice{\omega}{V} F_1(p) F_2(p)}{1-F_1(p) \indice{A}{V}\indice{\omega}{V} F_2(p) \left(k_1 +p k_2 - p\indice{A}{V}\indice{\omega}{v}\right)}$.

Et donc 
$H(p) = \dfrac{- \indice{A}{V}\indice{\omega}{V} \dfrac{1}{\indice{B}{V} p + k_3} \dfrac{1}{\left(\indice{I}{1y} +\indice{B}{V} \right)p^2- Mgh_2}}{1-\dfrac{1}{\indice{B}{V} p + k_3} \indice{A}{V}\indice{\omega}{V} \dfrac{1}{\left(\indice{I}{1y} +\indice{B}{V} \right)p^2- Mgh_2} \left(k_1 +p k_2 - p\indice{A}{V}\indice{\omega}{v}\right)}$

$= \dfrac{- \indice{A}{V}\indice{\omega}{V} }{\left(\indice{B}{V} p + k_3 \right) \left(\left(\indice{I}{1y} +\indice{B}{V} \right)p^2 - Mgh_2 \right)- \indice{A}{V}\indice{\omega}{V} \left(k_1 +p k_2 - p\indice{A}{V}\indice{\omega}{v}\right)}$


$= \dfrac{- \indice{A}{V}\indice{\omega}{V} }{
 p^3\left( \indice{B}{V} \left(\indice{I}{1y} +\indice{B}{V} \right) \right)
+p^2\left( k_3 \left(\indice{I}{1y} +\indice{B}{V} \right) \right) 
+p \left( - Mgh_2 \indice{B}{V} - \indice{A}{V}\indice{\omega}{v} k_2 + \indice{A}{V}^2\indice{\omega}{v}^2\right)
- \indice{A}{V}\indice{\omega}{v} k_1 - Mgh_2 k_3 
}$

$= \dfrac{ \indice{A}{V}\indice{\omega}{V} }{
 - p^3\left( \indice{B}{V} \left(\indice{I}{1y} +\indice{B}{V} \right) \right)
- p^2\left( k_3 \left(\indice{I}{1y} +\indice{B}{V} \right) \right) 
 + p \left(  Mgh_2 \indice{B}{V} + \indice{A}{V}\indice{\omega}{v} k_2 - \indice{A}{V}^2\indice{\omega}{v}^2\right)
+ \indice{A}{V}\indice{\omega}{v} k_1 + Mgh_2 k_3 
}$

On a alors :

$A\left(k_1,k_3\right)= \indice{A}{V}\indice{\omega}{v} k_1 + Mgh_2 k_3$,

$B\left(k_2\right) = Mgh_2 \indice{B}{V} + \indice{A}{V}\indice{\omega}{v} k_2 - \indice{A}{V}^2\indice{\omega}{v}^2$

$C\left(k_3\right) = k_3 \left(\indice{I}{1y} +\indice{B}{V} \right)$

$D = -  \indice{B}{V} \left(\indice{I}{1y} +\indice{B}{V} \right) $


\textbf{Signe à vérifier sur D.}

\end{corrige}
